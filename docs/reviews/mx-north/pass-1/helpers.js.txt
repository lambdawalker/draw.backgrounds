import {rect,line,path,polygon,ellipse,samples,grid,ridge,pine,flower,rosette} from '../../../geometry.js';
export {rect,line,path,polygon,ellipse,samples,grid,ridge,pine,flower,rosette};
export const quality={pass:1};
export function paper(c,phase,colors,ink){c.add(rect(0,0,1400,880,c.gradient(colors.map((v,i)=>[i*100/(colors.length-1),v]))));c.glow(950,180,650,420,'#ffffff',0.7);grid(c,{color:ink,opacity:.2,spacing:15,amplitude:38,phase});}
export function contours(c,ink,offset=0,amp=45){for(let k=0;k<34;k++){let y=570+k*8+offset;c.add(line(samples(x=>[x,y+amp*Math.sin(x/260+k*.033)+14*Math.sin(x/87+k*.08)],-30,1440,8),ink,.8,.27));}}
export function leaf(c,x,y,size,angle,ink){let s='';for(let k=-7;k<=7;k++)s+=path(`M0 0 Q${size*.55} ${k*size*.06} ${size} 0`,ink,.8,.42);for(let k=1;k<10;k++)s+=path(`M${k*size/10} 0 l${size*.08} ${size*.17*Math.sin(k/10*Math.PI)} M${k*size/10} 0 l${size*.08} ${-size*.17*Math.sin(k/10*Math.PI)}`,ink,.7,.4);c.add(`<g transform="translate(${x} ${y}) rotate(${angle})">${s}</g>`);}
export function agave(c,x,y,s,ink){for(let j=-6;j<=6;j++){let xx=x+j*s*.14,yy=y-s*(.4+.6*Math.cos(j/7));c.add(path(`M${x} ${y} Q${x+j*s*.035} ${y-s*.5} ${xx} ${yy} Q${x+j*s*.13} ${y-s*.19} ${x} ${y}`,ink,.9,.4,ink));for(let k=0;k<3;k++)c.add(path(`M${x+k} ${y} Q${x+j*s*.06+k} ${y-s*.4} ${xx} ${yy}`,ink,.7,.5));}}
export function cactus(c,x,y,s,ink){let d=`M${x-12*s} ${y} V${y-140*s} Q${x} ${y-165*s} ${x+12*s} ${y-140*s} V${y} Z M${x-12*s} ${y-67*s} Q${x-70*s} ${y-62*s} ${x-62*s} ${y-122*s} Q${x-53*s} ${y-139*s} ${x-45*s} ${y-122*s} V${y-92*s} L${x-12*s} ${y-92*s} M${x+12*s} ${y-100*s} H${x+39*s} V${y-154*s} Q${x+49*s} ${y-174*s} ${x+58*s} ${y-154*s} V${y-89*s} Q${x+50*s} ${y-75*s} ${x+12*s} ${y-77*s}`;c.add(path(d,ink,1,.36,ink));for(let j=-2;j<=2;j++)c.add(path(`M${x+j*4*s} ${y} V${y-138*s}`,ink,.7,.7));}
export function palm(c,x,y,s,ink){c.add(path(`M${x} ${y} Q${x-35*s} ${y-120*s} ${x} ${y-220*s}`,ink,4*s,.36));for(let j=0;j<9;j++)leaf(c,x,y-220*s,135*s,205+j*20,ink);}
export function arch(c,x,y,w,h,ink){c.add(path(`M${x-w} ${y} Q${x-w*.8} ${y-h} ${x} ${y-h} Q${x+w*.8} ${y-h} ${x+w} ${y} L${x+w*.5} ${y} Q${x+w*.45} ${y-h*.65} ${x} ${y-h*.6} Q${x-w*.45} ${y-h*.65} ${x-w*.5} ${y}Z`,ink,1,.32,ink));for(let j=0;j<24;j++){let t=Math.PI*j/23;c.add(line([[x+Math.cos(t)*w*.59,y-Math.sin(t)*h*.62],[x+Math.cos(t)*w*.95,y-Math.sin(t)*h*.95]],ink,.7,.4));}}
export function vine(c,x,y,ink){for(let k=0;k<6;k++){leaf(c,x+k*40,y-k*20,70,210+k*8,ink);for(let j=0;j<4;j++)c.add(ellipse(x+k*40+j%2*10,y-k*20+20+Math.floor(j/2)*10,8,9,ink,.18));}c.add(path(`M${x-40} ${y+60} Q${x+100} ${y-80} ${x+250} ${y-100}`,ink,1.3,.4));}
