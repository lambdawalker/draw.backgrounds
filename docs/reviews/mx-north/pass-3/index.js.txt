  paper(c, p, ["#fff8ed", "#eef7f4", "#f8e9eb"], "#aa9c8a");
  contours(c, "#729c9f", 60, 32);
  for (let i = 0; i < 7; i++) {
    arch(c, 650 + i * 95, 650, 48, 150, "#ad928d");
    c.add(rect(600 + i * 95, 495, 100, 10, "#ad928d", 0.24));
    for (let k = 0; k < 4; k++)
      c.add(
        line(
          [
            [610 + i * 95, 505 + k * 7],
            [690 + i * 95, 505 + k * 7],
          ],
          "#a38b89",
          0.7,
          0.27,
        ),
      );
  }
  vine(c, 1080, 760, "#92858d");
  rosette(c, 260, 680, 110, "#ad9d8f", { rings: 16, lobes: 12, opacity: 0.16 });
}
